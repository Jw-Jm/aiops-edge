```go
// buildAlertMessage 按规则类型生成语义正确的告警消息：
// threshold 保留"值 > 阈值"格式；anomaly 说明 zscore/MAD 检测结果；
// burn_rate 展示烧毁倍数；其余类型回退阈值格式。
func buildAlertMessage(rule AlertRule, value float64) string {
	switch rule.Type {
	case "anomaly":
		method := rule.AnomalyMethod
		if method == "" { method = "zscore" }
		return fmt.Sprintf("%s: %s 异常（%s=%.2f > 阈值 %.2f，偏离历史基线）",
			rule.Name, rule.Metric, method, value, rule.Threshold)
	case "burn_rate":
		return fmt.Sprintf("SLO 烧毁率告警: %s 烧毁率 %.1fx > 阈值 %.1fx",
			rule.Metric, value, rule.Threshold)
	default:
		return fmt.Sprintf("%s: %s %.2f > threshold %.2f", rule.Name, rule.Metric, value, rule.Threshold)
	}
}
```

调用处（原 `msg := fmt.Sprintf("%s: %s %.2f > threshold %.2f", ...)` 一行）替换为 `msg := buildAlertMessage(rule, value)`。

// detectTrendGaps 基于"近 n 小时应有点"检测 trend 缺失小时，返回缺口描述。
// rows 的 t 字段格式 "2006-01-02 15:04:05"。
func detectTrendGaps(rows []map[string]interface{}, hours int) []string {
	now := time.Now().Truncate(time.Hour)
	got := map[int64]bool{}
	for _, r := range rows {
		if ts, ok := r["t"].(string); ok {
			if t, err := time.ParseInLocation("2006-01-02 15:04:05", ts, time.Local); err == nil {
				got[t.Unix()] = true
			}
		}
	}
	var gaps []string
	var gapStart, gapEnd *time.Time
	flush := func() {
		if gapStart != nil {
			gaps = append(gaps, fmt.Sprintf("%s ~ %s", gapStart.Format("01-02 15:00"), gapEnd.Format("01-02 15:00")))
			gapStart, gapEnd = nil, nil
		}
	}
	for i := hours - 1; i >= 0; i-- {
		h := now.Add(-time.Duration(i) * time.Hour)
		if !got[h.Unix()] {
			if gapStart == nil { gapStart = &h }
			gapEnd = &h
		} else {
			flush()
		}
	}
	flush()
	return gaps
}
```

// DashboardResources handles GET /api/v1/dashboard/resources?cluster_id=X
// 工作台集群资源卡：CPU/内存/磁盘当前使用率 + 阈值 + 预计触达时间 + 节点数。
func (h *Handler) DashboardResources(w http.ResponseWriter, r *http.Request) {
	cid := r.URL.Query().Get("cluster_id")
	if cid == "" { cid = "all" }
	// 节点数：cluster_id=default 映射主集群（id=1），其余按 name 匹配
	nodeCount := 0
	if dao := &store.ClusterDAO{}; dao != nil {
		if clusters, err := dao.List(); err == nil {
			for _, c := range clusters {
				match := (cid == "default" && c.ID == 1) || c.Name == cid
				if match { nodeCount = c.NodeCount; break }
			}
		}
	}
	// 三指标查询（复用容量 PromQL，最多并行 3 次，各 10s 超时）
	type res struct{ metric, cur, threshold string; ett int }
	metrics := []struct{ m, thr string }{{"cpu", "80"}, {"memory", "90"}, {"disk", "85"}}
	items := []map[string]interface{}{}
	for _, mc := range metrics {
		start := time.Now().Add(-time.Hour).Unix()
		end := time.Now().Unix()
		promQL := capacityPromQLForCluster(mc.m, "", cid)
		if promQL == "" { continue }
		series, err := h.vmRangeQuery(promQL, start, end, 300)
		if err != nil || len(series) == 0 {
			items = append(items, map[string]interface{}{
				"metric": mc.m, "current": nil, "threshold": mc.thr, "ett_seconds": 0, "error": "no data",
			})
			continue
		}
		last := series[len(series)-1]
		thr, _ := strconv.ParseFloat(mc.thr, 64)
		ett := 0
		if last >= thr { ett = 0 } else if len(series) >= 2 {
			// 线性外推 ETT（复用容量预测思路的简化版）
			slope := (last - series[len(series)-2]) / float64(series[len(series)-1][0]-series[len(series)-2][0])
			if slope > 0 { ett = int((thr - last) / slope) }
		}
		items = append(items, map[string]interface{}{
			"metric": mc.m, "current": last, "threshold": thr, "ett_seconds": ett,
		})
	}
	respondJSON(w, http.StatusOK, map[string]interface{}{
		"cluster_id": cid, "node_count": nodeCount, "resources": items,
	})
}
```
